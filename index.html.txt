        let w = 40 + Math.min(1, ratio) * 280;
        return Math.min(320, Math.max(2, w));
    }

            }
            svg.appendChild(text);
        }

        // 底部效率说明
        let effNote = document.createElementNS("http://www.w3.org/2000/svg", "text");
        effNote.textContent = `↓ 当前传递效率: ${efficiency}%  ↓  约${(100 - efficiency).toFixed(1)}% 以呼吸热散失`;
        effNote.setAttribute("x", width/2);
        effNote.setAttribute("y", height - 12);
        effNote.setAttribute("text-anchor", "middle");
        effNote.setAttribute("font-size", "11");
        effNote.setAttribute("fill", "#4f6b4a");
        svg.appendChild(effNote);

        let titleNote = document.createElementNS("http://www.w3.org/2000/svg", "text");
        titleNote.textContent = "💡 宽度 = 相对能量比例 | 极细线表示能量＜0.2 kcal 或比例＜0.02%";
        titleNote.setAttribute("x", width - 20);
        titleNote.setAttribute("y", 20);
        titleNote.setAttribute("text-anchor", "end");
        titleNote.setAttribute("font-size", "10");
        titleNote.setAttribute("fill", "#5a6e4a");

    }

    // 更新右侧表格（绝对能量值）
    function updateEnergyTable() {
        updateEnergies();
        for (let i = 0; i < energies.length; i++) {
            let val = energies[i];
            let displayVal = val < 0.01 ? "<0.01" : val.toFixed(1);
            document.getElementById(`energy${i}`).innerHTML = displayVal + " kcal/m²·年";
        }
        
        // 更新表格中的物种信息
        const select = document.getElementById("sceneSelect");
        let selectedScene = select.value;
        let species = sceneSpecies[selectedScene];
        if(species) {
            document.getElementById("species0").innerHTML = species.producer;
            document.getElementById("species1").innerHTML = species.primary;
            document.getElementById("species2").innerHTML = species.secondary;
            document.getElementById("species3").innerHTML = species.top;
        } else {
            for(let i=0; i<4; i++) document.getElementById(`species${i}`).innerHTML = "--";
        }
    }

    // 显示营养级详情
    function showLevelDetail(levelIndex) {
        const level = trophicLevels[levelIndex];
        const energyAbs = energies[levelIndex];
        const percent = (energyAbs / energies[0] * 100).toFixed(3);
        const eff = efficiency;
        
        const select = document.getElementById("sceneSelect");
        let selectedScene = select.value;
        let speciesData = sceneSpecies[selectedScene];
        let speciesText = "";
        if(speciesData) {
            switch(levelIndex) {
                case 0: speciesText = speciesData.producer; break;
                case 1: speciesText = speciesData.primary; break;
                case 2: speciesText = speciesData.secondary; break;
                case 3: speciesText = speciesData.top; break;
                default: speciesText = "--";
            }
        }
        
        let extraMsg = "";
        if (levelIndex === 0) {
            extraMsg = ` 生产者固定的总能量为 ${producerEnergy} kcal/m²·年，是生态系统的起点。`;
        } else if (levelIndex === 3) {
            extraMsg = ` 顶级捕食者位于金字塔最顶层，获得的能量仅为 ${energyAbs < 0.01 ? '<0.01' : energyAbs.toFixed(2)} kcal/m²·年，占总能量的 ${percent}%。当前能量已接近耗竭${energyAbs < 0.2 ? '，金字塔显示为细线。' : '。'}`;
        } else {
            extraMsg = ` 该营养级获得的绝对能量为 ${energyAbs.toFixed(1)} kcal/m²·年，占生产者能量的 ${percent}%。传递效率为 ${eff}%。`;
        }
        
        const detailHtml = `<strong>${level.name} · ${level.trophic}</strong><br>
        📖 生态角色：${roleDescription[levelIndex]}<br>
        🌿 典型代表生物：${speciesText}<br>
        ⚡ 绝对能量值：${energyAbs < 0.01 ? '<0.01' : energyAbs.toFixed(2)} kcal/m²·年<br>
        📊 相对生产者能量比例：${percent}%<br>
        🔎 ${extraMsg}<br>
        🌟 应用：若该营养级生物增重1kg，至少需要上一营养级提供约 ${Math.ceil(1 / (efficiency/100))} kg 有机物。`;
        
        document.getElementById("detailText").innerHTML = detailHtml;
        const detailDiv = document.getElementById("detailBox");
        detailDiv.style.background = "#f7f3e0";
        setTimeout(() => { detailDiv.style.background = "#f4f1e1"; }, 200);
    }

    // 刷新所有UI
    function refreshAll() {
        updateEnergies();
        drawPyramid();
        updateEnergyTable();
        
        const select = document.getElementById("sceneSelect");
        let selectedValue = select.value;
        let scene = scenes[selectedValue];
        if (Math.abs(efficiency - scene.efficiency) < 0.2 && Math.abs(producerEnergy - scene.producerEnergy) < 50) {
            document.getElementById("sceneExplanation").innerHTML = `<strong>📖 科学解释</strong><br>${scene.explanation}`;
            document.getElementById("realDataDisplay").innerHTML = `📍 <strong>${selectedValue==='cedar'?'赛达伯格湖':(selectedValue==='qinglong'?'成都青龙湖':(selectedValue==='marine'?'热带海洋浮游链':'寒带针叶林'))} 数据参考</strong><br>${scene.dataRef}<br>🔬 传递效率: ${scene.efficiency}% | 生产者能量: ${scene.producerEnergy} kcal/m²·年<br>🌿 代表生物信息已同步至右侧表格。<br>📊 参考范围: ${scene.range}`;
            document.getElementById("energyRangeHint").innerHTML = `参考范围: ${scene.rangeMin} ~ ${scene.rangeMax} (${selectedValue==='cedar'?'温带湖泊':(selectedValue==='qinglong'?'成都平原湖泊':(selectedValue==='marine'?'近岸海洋/上升流区域':'寒带森林'))})`;
            document.getElementById("producerEnergyInput").min = scene.rangeMin;
            document.getElementById("producerEnergyInput").max = scene.rangeMax;
        } else {
            document.getElementById("sceneExplanation").innerHTML = `<strong>📖 自定义模式</strong><br>当前效率 ${efficiency}%，生产者总能量 ${producerEnergy} kcal/m²·年。你可以调整参数或选择预设场景对比。`;
            document.getElementById("realDataDisplay").innerHTML = `📍 <strong>自定义参数</strong><br>传递效率 ${efficiency}%，生产者总能量 ${producerEnergy} kcal/m²·年。各营养级能量见右侧表格。`;
            document.getElementById("energyRangeHint").innerHTML = `参考范围: 500 ~ 15000 (热带雨林可达15000)`;
            document.getElementById("producerEnergyInput").min = 500;
            document.getElementById("producerEnergyInput").max = 15000;
            if(!scenes[selectedValue] || (Math.abs(efficiency - scenes[selectedValue].efficiency) > 0.2 || Math.abs(producerEnergy - scenes[selectedValue].producerEnergy) > 50)) {
                for(let i=0; i<4; i++) document.getElementById(`species${i}`).innerHTML = "自定义场景，数据待补充";
            }
        }
    }

    // 设置场景
    function setScene(sceneId) {
        const scene = scenes[sceneId];
        if (!scene) return;
        efficiency = scene.efficiency;
        producerEnergy = scene.producerEnergy;
        document.getElementById("efficiencySlider").value = efficiency;
        document.getElementById("efficiencyValue").innerText = efficiency.toFixed(1);
        document.getElementById("producerEnergyInput").value = producerEnergy;
        refreshAll();
        document.getElementById("detailText").innerHTML = "点击任意一级营养级，可查看该营养级在所选场景中的典型代表生物和能量数据。";
    }

    // 绑定事件
    function bindEvents() {
        const slider = document.getElementById("efficiencySlider");
        slider.addEventListener("input", function() {
            efficiency = parseFloat(slider.value);
            document.getElementById("efficiencyValue").innerText = efficiency.toFixed(1);
            refreshAll();
        });
        const energyInput = document.getElementById("producerEnergyInput");
        energyInput.addEventListener("input", function() {
            let val = parseFloat(energyInput.value);
            if (!isNaN(val) && val >= 0) {
                producerEnergy = val;
                refreshAll();
            }
        });
        const sceneSelect = document.getElementById("sceneSelect");
        sceneSelect.addEventListener("change", function() {
            setScene(sceneSelect.value);
        });
    }

    // 初始化
    function init() {
        bindEvents();
        setScene("cedar");
    }
    init();
